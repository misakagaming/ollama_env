```java
        charSeq1.append(charSeq2);
        System.out.println(charSeq1);

class CharSeq {
    private char[] data;

    public CharSeq(String str) {
        this.data = str.toCharArray();
    }

    public void append(CharSeq other) {
        if (other == null || other.data == null) {
            return;
        }
        int newLength = this.data.length + other.data.length;
        char[] newData = new char[newLength];
        System.arraycopy(this.data, 0, newData, 0, this.data.length);
        System.arraycopy(other.data, 0, newData, this.data.length, other.data.length);
        this.data = newData;
    }

    @Override
    public String toString() {
        return new String(this.data);
    }
}
```

```csharp
    public static void CopyWithGap<K>(K[] source, K[] dest, int length, int gapIndex)

        Array.Copy(source, gapIndex, dest, gapIndex + 1, length - gapIndex);
    }

    internal readonly int length;
    internal readonly bool extendable;

    public const bool _BOOL = true;

    {
        this.array = array;
        this.length = length;

        this.canExtend = extendable ? new AtomicBool(false) : null;

    {
        int len = length + _NUM;
        int newLen = len;
        bool extendable;

        if (index == len - _NUM)
        {
            var x = canExtend;
            if (x != null)
            {

                if (array.Length > index && x.GetAndSet(_BOOL))
```

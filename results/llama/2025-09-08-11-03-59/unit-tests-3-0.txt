    {
        // Get the lengths of the input arrays
        int lengthX = x.Length;
        int lengthY = y.Length;

        // Find the minimum length
        int min = Math.Min(lengthX, lengthY);

        // Count bits in common between sets and set unique bits

    private readonly List<byte[]> mBuffersBySize = new List<byte[]>();

            mBuffersByLastUse.Add(buf);
            int pos = mBuffersBySize.BinarySearch(buf, new BufferSizeComparer());
            if (pos < 0) pos = ~pos;

    local.put("key1", "val1");
    local.put("key2", "localVal2");
    local.put("key3", "val3");

    Map<String, String> remote = new LinkedHashMap<>();

    for (String key : local.keySet()) {

        } else {
            merged.put(key, local.get(key));
        }

    assertEquals("val1", merged.get("key1"));

```csharp
namespace MissileExample
```

```csharp
using System;

public class DeltaMax
{
    private long _maxDelta = 0;

    public void Update(long delta)
    {
        if (delta > _maxDelta)
        {
            _maxDelta = delta;
        }
    }

    public long Get()
    {
        return _maxDelta;
    }
}

public class TimeDeltaTracker
{
    private DeltaMax deltaMax = new DeltaMax();

    public void TrackDelta(long delta)
    {
        deltaMax.Update(delta);
    }

    // get running - maximum time - delta , in nanoseconds .
    public long Max()
    {
        return deltaMax.Get();
    }
}

class Program
{
    static void Main(string[] args)
    {
        TimeDeltaTracker tracker = new TimeDeltaTracker();

        long[] runningTimes = {15000000L, 12750000L, 22300000L, 19800000L};
        foreach (long t in runningTimes)
        {
            tracker.TrackDelta(t);
        }

        Console.WriteLine("Maximum time delta in nanoseconds: " + tracker.Max());
    }
}
```
